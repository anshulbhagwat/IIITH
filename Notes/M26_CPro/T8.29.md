# Print abci

Given an input $n$, for each number from one to $n$, print:
1. `a` if the number is divisible by 6
2. `b` if the number is divisible by 7
3. `c` if the number is divisible by 6 and 7
4. the number itself if it is neither divisible by 6 nor 7



```C

#include <stdio.h>

int main() {
    int n;
    scanf("%d", &n);

    for(int i = 1; i<= n; i++){ // 1, 2, 3, 4, 5, .... , n
        if (i % 6 == 0 && i % 7 == 0){ // if i is divisible by 6 AND i is divisible by 7
            printf("c ");
            continue;
        }
        if (i % 6 == 0){ // if i is divisible by 6 but NOT by 7
            printf("a ");
            continue;
        }
        if (i % 7 == 0){ // if i is divisible by 7 but NOT by 6
            printf("b ");
            continue;
        }
        // otherwise
        printf("%d ", i);
    }    

    return 0;
}
```

# Unique Number
There are $n$ numbers where each number is paired with another number, except for one. Using only bitwise operators, find the one number which does not occur twice in the $n$ numbers.

Input Format:
```
n
num_1
num_2
num_3
.
.
.
num_n
```

Output Format:
```
num
```


Property of XOR
1. `A ^ 0 = A`
2. `A ^ A = 0`


```
1 ^ 1 = 0
0 ^ 1 = 1
1 ^ 0 = 1
0 ^ 0 = 0
```


```C
#include <stdio.h>

int main() {
    int n, num, answer = 0;
    scanf("%d", &n);

    for(int i = 1; i<= n; i++){ // 1, 2, 3, 4, 5, .... , n
        scanf("%d", &num);
        answer = answer ^ num;

        // printf("%b %b\n", num, answer);
    }    
    printf("%d", answer);

    return 0;
}
```

# Calculator
Write a program which keeps running in loops. In each loop, first take a user input '+', '-', '/', '\*', or 'Q'.
Based on this, take in two more integers and perform the operation. Print the output. If the operation is division and the second argument is 0, print "ERROR: DIVISION BY ZERO".
And 'Q' is quit.

```C
#include <stdio.h>

int main() {
    int n;

    char op;
    int a, b;
    scanf("%d", &n);
    while(1){
        scanf("%c", &op);
        if (op == 'Q'){
            break;
        }

        
        if (op == '+'){
            scanf("%d%d", &a, &b);
            printf("%d", a+b);
            continue;
        }
        
        if (op == '-'){
            scanf("%d%d", &a, &b);
            printf("%d", a-b);
        }
        
        if (op == '*'){
            scanf("%d%d", &a, &b);
            printf("%d", a*b);
        }
        
        if (op == '/'){
            scanf("%d%d", &a, &b);
            if (b)
                printf("%d", a/b);

            else
                printf("ERROR: DIVISION BY ZERO");
        }
    }
    
    
    return 0;
}
```
