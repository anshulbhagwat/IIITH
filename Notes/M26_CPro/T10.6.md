```C
void sortColors(int* nums, int numsSize) {
    int n = numsSize;
    int c0 = 0, c1 = 0, c2 = 0;

    for(int i = 0; i<n; i++){
        if(nums[i] == 0)
            c0++;
        if(nums[i] == 1)
            c1++;
        if(nums[i] == 2)
            c2++;       
    }
    int i = 0;
    while(c0){
        nums[i++] = 0;
        c0--;
    }
        
    while(c1){
        nums[i++] = 1;
        c1--;
    }

    while(c2){
        nums[i++] = 2;
        c2--;
    }

}
```

```C

#include <stdio.h>

int main() {
    int n, x;
    scanf("%d%d", &n, &x);
    int a[n];
    for(int i = 0; i<n; i++)
        scanf("%d", &a[i]);

    for(int i = 0; i<n; i++){
        int mi = i;
        for(int j = i+1; j<n; j++){
            if (a[mi] > a[j])
                mi = j;
        }
        int temp = a[mi];
        a[mi] = a[i];
        a[i] = temp;       
    }
    int l = 0, r = n-1, c = 0;
    // c is the count for gondolas
    while(l <= r){
        if (a[l] + a[r] <= x){
            // this is the case where two children are paired
            l++;
            r--;
            c++;
            continue;
        }
        // this is the case where a child is alone
        r--;
        c++;
    }

    printf("%d", c);

    return 0;
}
```
