strcmp
```C
#include <stdio.h>
#include <string.h>


int main() {
    char lhs[100] = "apple";
    char rhs[100] = "apple";
    int n1 = strlen(lhs);
    int n2 = strlen(rhs);
    int n = n1;
    if (n2 < n){
        n = n2;
    }
    
    for(int i = 0; i < n; i++){
        if(lhs[i] < rhs[i]){
            printf("-1");
            return 0;
        }
        if(lhs[i] > rhs[i]){
            printf("1");
            return 0;
        }
    }
    if(n1 < n2){
        printf("-1");
        return 0;
    }
    if(n2 < n1){
        printf("1");
        return 0;
    }
    printf("0");
    return 0;
}
```



Sort An Array of Words into Dictionary Order
```C
#include <stdio.h>
#include <string.h>


int main() {
    int n;
    scanf("%d", &n);
    char words[n][100];

    for (int i = 0; i < n; i++){
        scanf("%s", words[i]);
    }

    for (int i = 0; i<n-1; i++){
        for(int j = 0; j<n-i; j++){
            int x = strcmp(words[j], words[j+1]);
            char temp[100];
            if (x > 0){
                strcpy(temp, words[j]);
                strcpy(words[j], words[j+1]);
                strcpy(words[j+1], temp);
            }
        }
    }

    for(int i = 0; i<n; i++){
        printf("%s\n", words[i]);
    }
    
    return 0;
}
```
